Представим что это наш файл с исходным кодом проекта.

git config --list // посмотреть текущие значения конфига GIT  

git config --global user.name "Maksim Deev" // Установить глобальную конфигурацию имя пользователя
git config --global user.email "maks@mail.com" // Установить глобальную конфигурацию email

Две эти конфигурационные настройки нелбходимо обязательно установить перед первым использованием git


git init // Инициализация репозитория

git status // Показывает статус изменений проекта

git add *путь до файла* // Готовит файл к фиксации
git add . // подготовить все измененные и неотслеживаемые файлы к фиксации

git commit -m "сообщение коммита" // фиксация изменений

git log // просмотр лога коммитов
git show *хеш коммита* // просмотр полной информации по коммиту


git checkout *хеш* // переключение состояния проекта

git stash // спрятать незакоммиченные изменения
git stash pop // достать спрятанные изменения

git checkout master // переключение состояния проекта на последний коммит ветки

git checkout -b *название ветки* // создать новую ветку и переключиться на нее 
git checkout *название ветки* // переключиться на существующую ветку

git remote add origin *ссылка на репо* // создать ссылку на репо верхнего уровня
git pull origin master // получить все изменения ветки мастер
git push orgin master // отправить изменения из ветки мастер в origin
